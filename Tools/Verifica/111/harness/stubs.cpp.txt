static std::vector<std::string> fontNames;
int idDeviceContext::FindFont(const char *name) {
	for (size_t i = 0; i < fontNames.size(); i++) if (fontNames[i] == name) return (int)i + 1;
	fontNames.push_back(name); return (int)fontNames.size();
}
