static std::string lower(std::string s) { for (size_t i = 0; i < s.size(); i++) s[i] = (char)tolower((unsigned char)s[i]); return s; }

static std::string esc(const char *s, int n) {
	std::string r;
	for (int i = 0; i < n; i++) {
		unsigned char c = (unsigned char)s[i];
		char b[8];
		if (c == '\\') r += "\\\\";
		else if (c == '\n') r += "\\n";
		else if (c == '\r') r += "\\r";
		else if (c == '\t') r += "\\t";
		else if (c < 32 || c > 126) { sprintf(b, "\\x%02X", c); r += b; }
		else r += (char)c;
	}
	return r;
}
static std::string esc(const char *s) { return esc(s, (int)strlen(s)); }

idFile *idFileSystem::OpenFileRead(const char *n) {
	std::map<std::string, MemFile>::iterator it = files.find(norm(n));
	if (it == files.end()) return 0;
	idFile *f = new idFile; f->rd = it->second.text; f->name = n; f->ts = it->second.ts; return f;
}

int idFileSystem::ReadFile(const char *n, void **b, ID_TIME_T *ts) {
	std::map<std::string, MemFile>::iterator it = files.find(norm(n));
	if (it == files.end()) { if (b) *b = 0; return -1; }
	if (ts) *ts = it->second.ts;
	if (b) {
		char *buf = (char *)malloc(it->second.text.size() + 1);
		memcpy(buf, it->second.text.data(), it->second.text.size()); buf[it->second.text.size()] = 0;
		*b = buf;
	}
	return (int)it->second.text.size();
}

static void dumpWindow(idWindow *w, bool simple, int level) {
	std::string pad(level * 2, ' ');
	if (simple) {
		printf("%sS %s flags=%08X\n", pad.c_str(), w->name.c_str(), w->flags);
		printf("%s rect=%s back=%s fore=%s mat=%s border=%s\n", pad.c_str(), rc(w->rect).c_str(), v4(w->backColor).c_str(), v4(w->foreColor).c_str(), v4(w->matColor).c_str(), v4(w->borderColor).c_str());
		printf("%s tscale=%s rot=%s vis=%d text=\"%s\" bg=\"%s\"\n", pad.c_str(), fb(w->textScale).c_str(), fb(w->rotate).c_str(), (int)(bool)w->visible, esc(w->text.c_str()).c_str(), esc(w->backGroundName.c_str()).c_str());
		return;
	}
	int mask = 0;
	for (int i = 0; i < idWindow::SCRIPT_COUNT; i++) if (w->scripts[i]) mask |= 1 << i;
	printf("%sW %s flags=%08X simple=0 child=%d draw=%d scr=%03X tl=%d named=%d tr=%d upd=%d cmd=\"%s\"\n", pad.c_str(), w->name.c_str(), w->flags, w->children.Num(), w->drawWindows.Num(), mask,
		w->timeLineEvents.Num(), w->namedEvents.Num(), w->transitions.Num(), w->updateVars.Num(), esc(w->cmd.c_str()).c_str());
	printf("%s rect=%s back=%s fore=%s mat=%s border=%s hover=%s\n", pad.c_str(), rc(w->rect).c_str(), v4(w->backColor).c_str(), v4(w->foreColor).c_str(), v4(w->matColor).c_str(), v4(w->borderColor).c_str(), v4(w->hoverColor).c_str());
	printf("%s tscale=%s rot=%s vis=%d noev=%d notime=%d text=\"%s\" bg=\"%s\"\n", pad.c_str(), fb(w->textScale).c_str(), fb(w->rotate).c_str(), (int)(bool)w->visible, (int)(bool)w->noEvents, (int)(bool)w->noTime, esc(w->text.c_str()).c_str(), esc(w->backGroundName.c_str()).c_str());
	for (int i = 0; i < w->definedVars.Num(); i++) printf("%s var %s = %s\n", pad.c_str(), w->definedVars[i]->GetName(), w->definedVars[i]->c_str());
	if (idChoiceWindow *c = dynamic_cast<idChoiceWindow *>(w)) printf("%s X choice n=%d v=%d cur=%d\n", pad.c_str(), c->choices.Num(), c->values.Num(), c->currentChoice);
	if (idSliderWindow *s = dynamic_cast<idSliderWindow *>(w)) printf("%s X slider low=%s high=%s step=%s\n", pad.c_str(), fb(s->low).c_str(), fb(s->high).c_str(), fb(s->stepSize).c_str());
	for (int i = 0; i < w->drawWindows.Num(); i++) {
		if (w->drawWindows[i].win) dumpWindow(w->drawWindows[i].win, false, level + 1);
		else {
			// a simple window: the data of idSimpleWindow
			idSimpleWindow *s = w->drawWindows[i].simp;
			std::string p2((level + 1) * 2, ' ');
			printf("%sS %s flags=%08X\n", p2.c_str(), s->name.c_str(), s->flags);
			printf("%s rect=%s back=%s fore=%s mat=%s border=%s\n", p2.c_str(), rc(s->rect).c_str(), v4(s->backColor).c_str(), v4(s->foreColor).c_str(), v4(s->matColor).c_str(), v4(s->borderColor).c_str());
			printf("%s tscale=%s rot=%s vis=%d text=\"%s\" bg=\"%s\"\n", p2.c_str(), fb(s->textScale).c_str(), fb(s->rotate).c_str(), (int)(bool)s->visible, esc(s->text.c_str()).c_str(), esc(s->backGroundName.c_str()).c_str());
		}
	}
}

static std::vector<std::string> splitWords(const std::string &s) {
	std::vector<std::string> r; std::istringstream is(s); std::string w; while (is >> w) r.push_back(w); return r;
}
static std::string unesc(const std::string &s) {
	std::string r;
	for (size_t i = 0; i < s.size(); i++) {
		if (s[i] == '\\' && i + 1 < s.size()) { i++; r += (s[i] == 'n') ? '\n' : (s[i] == 't') ? '\t' : s[i]; }
		else r += s[i];
	}
	return r;
}

// a font of the harness: the metrics come from formulas that the Delphi test repeats (TGuiTestFonts)
static std::string genFontData(int p) {
	std::string d;
	for (int i = 0; i < 256; i++) {
		int height = 6 + p / 4 + (i % 5), top = height - 2 + (i % 3), bottom = i % 2, pitch = 8, xSkip = 4 + p / 6 + (i % 7);
		int iw = xSkip + 2, ih = height + 1, zero = 0;
		float s = (float)(i % 16) / 16.0f, t = (float)(i / 16) / 16.0f, s2 = s + 0.0625f, t2 = t + 0.0625f;
		d.append((const char *)&height, 4); d.append((const char *)&top, 4); d.append((const char *)&bottom, 4); d.append((const char *)&pitch, 4);
		d.append((const char *)&xSkip, 4); d.append((const char *)&iw, 4); d.append((const char *)&ih, 4);
		d.append((const char *)&s, 4); d.append((const char *)&t, 4); d.append((const char *)&s2, 4); d.append((const char *)&t2, 4); d.append((const char *)&zero, 4);
		char sh[32]; memset(sh, 0, 32); sprintf(sh, "xxxxxxg%d_%d", p, i); d.append(sh, 32);
	}
	float gs = 48.0f / p; d.append((const char *)&gs, 4);
	char nm[64]; memset(nm, 0, 64); sprintf(nm, "fonts/english/fontImage_%d.dat", p); d.append(nm, 64);
	return d;
}
static void addFontSet(const std::string &dir) {
	int sizes[3] = { 12, 24, 48 };
	for (int k = 0; k < 3; k++) { char n[256]; sprintf(n, "%s/fontImage_%d.dat", dir.c_str(), sizes[k]); MemFile m; m.text = genFontData(sizes[k]); m.ts = ++clockTicks; files[norm(n)] = m; }
}
static std::string unescText(const std::string &s) {
	std::string r;
	for (size_t i = 0; i < s.size(); i++) {
		if (s[i] == '\\' && i + 1 < s.size()) { i++; r += (s[i] == 'n') ? '\n' : (s[i] == 't') ? '\t' : (s[i] == 'r') ? '\r' : s[i]; }
		else r += s[i];
	}
	return r;
}
static float F(const std::string &s) { return (float)atof(s.c_str()); }
static idVec4 col(const std::vector<std::string> &w, size_t i) { return idVec4(F(w[i]), F(w[i + 1]), F(w[i + 2]), F(w[i + 3])); }
static std::string restFrom(const std::string &line, size_t words) {
	size_t pos = 0;
	for (size_t k = 0; k < words; k++) {
		while (pos < line.size() && line[pos] == ' ') pos++;
		while (pos < line.size() && line[pos] != ' ') pos++;
	}
	if (pos < line.size()) pos++;
	return line.substr(pos < line.size() ? pos : line.size());
}

// the operations on the device context: "dc <op> ..."
static void dcOp(const std::vector<std::string> &w, const std::string &rest) {
	idDeviceContext &dc = uiManagerLocal.dc;
	const std::string &op = w[1];
	if (op == "size") dc.SetSize(F(w[2]), F(w[3]));
	else if (op == "font") dc.SetFont(atoi(w[2].c_str()));
	else if (op == "findfont") printf("F %d\n", dc.FindFont(w[2].c_str()));
	else if (op == "fontset") addFontSet(w[2]);
	else if (op == "clip") dc.PushClipRect(F(w[2]), F(w[3]), F(w[4]), F(w[5]));
	else if (op == "popclip") dc.PopClipRect();
	else if (op == "clipping") dc.EnableClipping(atoi(w[2].c_str()) != 0);
	else if (op == "overstrike") dc.SetOverStrike(atoi(w[2].c_str()) != 0);
	else if (op == "tick") com_ticNumber = atoi(w[2].c_str());
	else if (op == "setcursor") dc.SetCursor(atoi(w[2].c_str()));
	else if (op == "transform") {
		idVec3 o(F(w[2]), F(w[3]), F(w[4]));
		idMat3 m(F(w[5]), F(w[6]), F(w[7]), F(w[8]), F(w[9]), F(w[10]), F(w[11]), F(w[12]), F(w[13]));
		dc.SetTransformInfo(o, m);
	}
	else if (op == "filled") dc.DrawFilledRect(F(w[2]), F(w[3]), F(w[4]), F(w[5]), col(w, 6));
	else if (op == "rect") dc.DrawRect(F(w[2]), F(w[3]), F(w[4]), F(w[5]), F(w[6]), col(w, 7));
	else if (op == "material") dc.DrawMaterial(F(w[3]), F(w[4]), F(w[5]), F(w[6]), declManager->FindMaterial(w[2].c_str()), col(w, 7), F(w[11]), F(w[12]));
	else if (op == "matrot") dc.DrawMaterialRotated(F(w[3]), F(w[4]), F(w[5]), F(w[6]), declManager->FindMaterial(w[2].c_str()), col(w, 7), F(w[11]), F(w[12]), F(w[13]));
	else if (op == "matrect") dc.DrawMaterialRect(F(w[3]), F(w[4]), F(w[5]), F(w[6]), F(w[7]), declManager->FindMaterial(w[2].c_str()), col(w, 8));
	else if (op == "cursor") { float x = F(w[2]), y = F(w[3]); dc.DrawCursor(&x, &y, F(w[4])); printf("K %s %s\n", fb(x).c_str(), fb(y).c_str()); }
	else if (op == "coords") { float x = F(w[2]), y = F(w[3]), ww = F(w[4]), h = F(w[5]); bool r = dc.ClippedCoords(&x, &y, &ww, &h, NULL, NULL, NULL, NULL); printf("O %d %s %s %s %s\n", (int)r, fb(x).c_str(), fb(y).c_str(), fb(ww).c_str(), fb(h).c_str()); }
	else if (op == "coordst") { float x = F(w[2]), y = F(w[3]), ww = F(w[4]), h = F(w[5]), s1 = F(w[6]), t1 = F(w[7]), s2 = F(w[8]), t2 = F(w[9]); bool r = dc.ClippedCoords(&x, &y, &ww, &h, &s1, &t1, &s2, &t2); printf("O %d %s %s %s %s %s %s %s %s\n", (int)r, fb(x).c_str(), fb(y).c_str(), fb(ww).c_str(), fb(h).c_str(), fb(s1).c_str(), fb(t1).c_str(), fb(s2).c_str(), fb(t2).c_str()); }
	else if (op == "text") {
		// dc text scale align r g b a x y w h wrap cursor calc limit <text>
		idList<int> breaks;
		idRectangle r(F(w[8]), F(w[9]), F(w[10]), F(w[11]));
		int n = dc.DrawText(unescText(restFrom(rest, 16)).c_str(), F(w[2]), atoi(w[3].c_str()), col(w, 4), r, atoi(w[12].c_str()) != 0, atoi(w[13].c_str()), atoi(w[14].c_str()) != 0, &breaks, atoi(w[15].c_str()));
		printf("T %d breaks", n); for (int i = 0; i < breaks.Num(); i++) printf(" %d", breaks[i]); printf("\n");
	}
	else if (op == "textwidth") printf("W %d\n", dc.TextWidth(unescText(restFrom(rest, 4)).c_str(), F(w[2]), atoi(w[3].c_str())));
	else if (op == "textheight") printf("H %d\n", dc.TextHeight(unescText(restFrom(rest, 4)).c_str(), F(w[2]), atoi(w[3].c_str())));
	else if (op == "charwidth") printf("CW %d\n", dc.CharWidth((char)atoi(w[3].c_str()), F(w[2])));
	else if (op == "maxchar") printf("MC %d %d\n", dc.MaxCharWidth(F(w[2])), dc.MaxCharHeight(F(w[2])));
	else if (op == "scrollbar") printf("SB %s\n", dc.GetScrollBarImage(atoi(w[2].c_str())) ? "ok" : "none");
}

static std::map<std::string, std::string> defaultCvars;

int main(int argc, char **argv) {
	setvbuf(stdout, 0, _IONBF, 0);
	idLib::common = &theCommon; idLib::fileSystem = &theFs; common = &theCommon; fileSystem = &theFs;
	if (argc < 2) return 2;
	cvarTable()["sys_lang"] = "english"; defaultCvars = cvarTable();
	std::ifstream in(argv[1], std::ios::binary);
	std::string line, curFile;
	std::vector<std::string> ops;
	std::string caseName;
	idUserInterfaceLocal *gui = 0;
	bool inCase = false;
	std::map<std::string, MemFile> caseFiles;
	std::string fileText; bool inFile = false; std::string fileName;
	while (std::getline(in, line)) {
		if (!line.empty() && line[line.size() - 1] == '\r') line.erase(line.size() - 1);
		if (line.compare(0, 6, "@case ") == 0) {
			caseName = line.substr(6); printf("## %s\n", caseName.c_str());
			uiManagerLocal.dc.Shutdown(); files.clear(); cvarTable() = defaultCvars; matTable().clear(); keysDown().clear(); com_ticNumber = 0;
			addFontSet("fonts/english"); uiManagerLocal.dc.Init(); gui = 0; inCase = true; inFile = false; continue;
		}
		if (line == "@end") { if (inFile) { MemFile m; m.text = fileText; m.ts = ++clockTicks; files[norm(fileName.c_str())] = m; inFile = false; } inCase = false; continue; }
		if (line.compare(0, 6, "@file ") == 0) {
			if (inFile) { MemFile m; m.text = fileText; m.ts = ++clockTicks; files[norm(fileName.c_str())] = m; }
			fileName = line.substr(6); fileText = ""; inFile = true; continue;
		}
		if (line.compare(0, 4, "@op ") == 0) {
			if (inFile) { MemFile m; m.text = fileText; m.ts = ++clockTicks; files[norm(fileName.c_str())] = m; inFile = false; }
			std::string rest = line.substr(4);
			printf("> %s\n", rest.c_str());
			std::vector<std::string> w = splitWords(rest);
			try {
				if (w[0] == "load") {
					gui = new idUserInterfaceLocal;
					gui->InitFromFile(w[1].c_str());
				} else if (w[0] == "frame") {
					gui->SetTime(atoi(w[1].c_str()));
					gui->GetDesktop()->RunTimeEvents(atoi(w[1].c_str()));
				} else if (w[0] == "state") {
					std::string key = w[1]; std::string val = unesc(rest.size() > 6 + key.size() + 1 ? rest.substr(6 + key.size() + 1) : std::string());
					gui->SetStateString(key.c_str(), val.c_str());
				} else if (w[0] == "statechanged") {
					gui->StateChanged(atoi(w[1].c_str()), false);
				} else if (w[0] == "named") {
					gui->HandleNamedEvent(w[1].c_str());
				} else if (w[0] == "activate") {
					const char *r = gui->Activate(atoi(w[1].c_str()) != 0, atoi(w[2].c_str()));
					printf("act \"%s\"\n", esc(r).c_str());
				} else if (w[0] == "trigger") {
					gui->Trigger(atoi(w[1].c_str()));
				} else if (w[0] == "dc") {
					dcOp(w, rest);
				} else if (w[0] == "cvar") {
					cvarTable()[w[1]] = (w.size() > 2) ? w[2] : "";
				} else if (w[0] == "keydown") {
					keysDown().insert(atoi(w[1].c_str()));
				} else if (w[0] == "keyup") {
					keysDown().erase(atoi(w[1].c_str()));
				} else if (w[0] == "mat") {
					MatInfo mi; mi.w = atoi(w[2].c_str()); mi.h = atoi(w[3].c_str()); mi.cine = atoi(w[4].c_str()); mi.defaulted = atoi(w[5].c_str()) != 0;
					matTable()[w[1]] = mi;
				} else if (w[0] == "dump") {
					printf("I interactive=%d pending=\"%s\"\n", (int)gui->IsInteractive(), esc(gui->GetPendingCmd().c_str()).c_str());
					dumpWindow(gui->GetDesktop(), false, 0);
				}
			} catch (idException &e) {
				printf("ERR %s\n", esc(e.error).c_str());
			}
			continue;
		}
		if (inFile) { fileText += line; fileText += "\n"; }
	}
	return 0;
}
