# Casi di Tests\Data\TextureDefs: una sezione per caso.
# Grammatica (vedi Tests\Source\DelphiDoom.Tests.TextureDefs.Corpus.pas):
#   format standard|strife     layout dei lump TEXTUREx (default standard)
#   kind textures|flats        che cosa prova il caso (default textures)
#   pnames N1 N2 ...           lump PNAMES ('~' = spazio)
#   tex1|tex2|tex3 NOME L A MASKED idx,ox,oy ...   una texture del lump (nessun patch = nessun gruppo)
#   empty2|empty3              il lump esiste ma e' vuoto
#   badoffset1 N               la voce N della directory di TEXTURE1 punta oltre la fine
#   pk3 NOME L A CONTAPATCH ox oy                  una voce di TEXTURE0
#   flats N1 N2 ...            i flat del motore (kind flats)
#   text TESTO                 testo di FLATINFO ('|' = CR LF)

=== basic
pnames WALL1 WALL2 DOOR
tex1 STARTAN 128 128 0 0,0,0
tex1 BIGDOOR 128 96 0 1,0,0 2,64,0
tex1 STARGATE 64 72 1 2,0,-8

=== lowercase-names
pnames wall1 pAtCh
tex1 sTaRtAn 64 64 0 0,0,0 1,32,32

=== padded-patch-names
pnames ~AB~ ~~C
tex1 T1 16 16 0 0,0,0 1,0,0

=== eight-characters
pnames ABCDEFGH
tex1 LONGNAME 256 128 0 0,0,0

=== three-lumps
pnames P0 P1 P2 P3
tex1 ONE 64 64 0 0,0,0
tex1 ONEB 64 64 0 1,0,0
tex2 TWO 128 64 0 2,0,0
tex3 THREE 32 32 0 3,-4,5
tex3 THREEB 32 32 0 3,0,0 0,16,16

=== second-lump-empty
pnames P0
tex1 ONE 64 64 0 0,0,0
empty2
tex3 THREE 8 8 0 0,0,0

=== empty-first-lump
pnames P0

=== no-patches

tex1 EMPTYTEX 64 64 0
tex1 AFTER 64 64 0 0,0,0

=== many-patches

=== extreme-values
