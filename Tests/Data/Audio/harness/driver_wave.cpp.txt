static std::string esc(const char *s, int n) {
	std::string r;
	for (int i = 0; i < n; i++) {
		unsigned char c = (unsigned char)s[i]; char b[8];
		if (c == '\\') r += "\\\\"; else if (c == '\n') r += "\\n"; else if (c == '\r') r += "\\r"; else if (c == '\t') r += "\\t";
		else if (c < 32 || c > 126) { sprintf(b, "\\x%02X", c); r += b; } else r += (char)c;
	}
	return r;
}
static std::string esc(const char *s) { return esc(s, (int)strlen(s)); }

static void putLE(std::string &s, unsigned long v, int n) { for (int i = 0; i < n; i++) s += (char)((v >> (8 * i)) & 0xFF); }

static unsigned long number(const std::string &s) { return strtoul(s.c_str(), 0, 0); }

// @bytes tokens: T:abcd (four characters), B:n, W:n (16 bits), D:n (32 bits), Z:n (n zero bytes), X:hex
static void addBytes(std::string &out, const std::string &line) {
	std::stringstream ss(line); std::string w;
	while (ss >> w) {
		std::string v = w.substr(2);
		if (w[0] == 'T') { for (size_t i = 0; i < v.size(); i++) out += (v[i] == '_' ? ' ' : v[i]); }
		else if (w[0] == 'B') putLE(out, number(v), 1);
		else if (w[0] == 'W') putLE(out, number(v), 2);
		else if (w[0] == 'D') putLE(out, number(v), 4);
		else if (w[0] == 'Z') out += std::string((size_t)number(v), '\0');
		else if (w[0] == 'X') for (size_t i = 0; i + 1 < v.size(); i += 2) out += (char)strtoul(v.substr(i, 2).c_str(), 0, 16);
	}
}

int main(int argc, char **argv) {
	setvbuf(stdout, NULL, _IONBF, 0);
	common = &theCommon; idLib::common = &theCommon; fileSystem = &theFs; idLib::fileSystem = &theFs;
	std::ifstream in(argv[1]);
	std::string line, curFile;
	bool inFile = false;
	while (std::getline(in, line)) {
		if (!line.empty() && line[line.size() - 1] == '\r') line.erase(line.size() - 1);
		if (line.compare(0, 6, "@case ") == 0) { files.clear(); inFile = false; printf("## %s\n", line.substr(6).c_str()); continue; }
		if (line == "@end") { inFile = false; continue; }
		if (line.compare(0, 6, "@file ") == 0) { curFile = line.substr(6); files[curFile] = ""; inFile = true; continue; }
		if (line.compare(0, 4, "@op ") == 0) {
			inFile = false;
			std::stringstream ss(line.substr(4)); std::vector<std::string> t; std::string w;
			while (ss >> w) t.push_back(w);
			handleOp(t);
			continue;
		}
		if (inFile) addBytes(files[curFile], line);
	}
	return 0;
}
