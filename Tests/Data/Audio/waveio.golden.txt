## read and seek in a wave file
> open a.wav
ret=0
 size=10 mem=20 base=44 cksize=20
> read 8
read ret=8 got=8 hash=77043D3D
> read 4
read ret=4 got=4 hash=6C1014A5
> seek 6
seek ret=0
> read 4
read ret=4 got=4 hash=44B1B7B1
> seek 100
seek ret=0
> read 4
read ret=0 got=0 hash=811C9DC5
> reset
reset ret=0
> read 100
read ret=20 got=20 hash=3109CEC5
> read 100
read ret=0 got=0 hash=811C9DC5
> seek 0
seek ret=0
> read 3
read ret=3 got=3 hash=C1A2B2AA
