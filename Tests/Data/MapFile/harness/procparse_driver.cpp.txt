static void F(float f) { unsigned u; memcpy(&u, &f, 4); printf(" #%08x", u); }

static void Case(const char *name, const char *text, bool portals) {
	idLexer lex(LEXFL_NOSTRINGCONCAT | LEXFL_NOSTRINGESCAPECHARS | LEXFL_ALLOWPATHNAMES);
	lex.LoadMemory(text, (int)strlen(text), name);
	idRenderWorldLocal w;
	try {
		if (portals) w.ParseInterAreaPortals(&lex); else w.ParseNodes(&lex);
	} catch (ErrorThrown &) { printf("PROC %s error |\n", name); return; }
	if (!portals) {
		printf("PROC %s nodes | %d\n", name, w.numAreaNodes);
		for (int i = 0; i < w.numAreaNodes; i++) {
			printf("PROC %s n%d |", name, i);
			for (int k = 0; k < 4; k++) F(w.areaNodes[i].plane[k]);
			printf(" %d %d\n", w.areaNodes[i].children[0], w.areaNodes[i].children[1]);
		}
		return;
	}
	printf("PROC %s portals | %d %d\n", name, w.numPortalAreas, w.numInterAreaPortals);
	for (int i = 0; i < w.numInterAreaPortals; i++) for (int s = 0; s < 2; s++) {
		portal_t *p = w.doublePortals[i].portals[s];
		printf("PROC %s p%d_%d | %d %d", name, i, s, p->intoArea, p->w->GetNumPoints());
		for (int j = 0; j < p->w->GetNumPoints(); j++) { F((*p->w)[j][0]); F((*p->w)[j][1]); F((*p->w)[j][2]); }
		printf("\nPROC %s p%d_%d plane |", name, i, s);
		for (int k = 0; k < 4; k++) F(p->plane[k]);
		printf("\n");
	}
	for (int a = 0; a < w.numPortalAreas; a++) {
		int n = 0; for (portal_t *p = w.portalAreas[a].portals; p; p = p->next) n++;
		printf("PROC %s a%d | %d", name, a, n);
		for (portal_t *p = w.portalAreas[a].portals; p; p = p->next) printf(" %d", p->intoArea);
		printf("\n");
	}
}
