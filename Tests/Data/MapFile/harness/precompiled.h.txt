struct Blob {
	char *p; int n;
	Blob() : p(0), n(0) {}
	void Append(const char *s, int l) { p = (char *)realloc(p, n + l + 1); memcpy(p + n, s, l); n += l; p[n] = 0; }
};
class idFile {
public:
	Blob rd; int pos; char name[512]; Blob out; long ts;
	idFile() : pos(0), ts(0) { name[0] = 0; }
	int Length() { return rd.n; }
	int Read(void *b, int l) { if (pos + l > rd.n) l = rd.n - pos; memcpy(b, rd.p + pos, l); pos += l; return l; }
	int Write(const void *b, int l) { out.Append((const char *)b, l); return l; }
	const char *GetFullPath() { return name; }
	long Timestamp() { return ts; }
	int WriteFloatString(const char *fmt, ...) { char buf[16384]; va_list ap; va_start(ap, fmt); int len = FS_WriteFloatString(buf, fmt, ap); va_end(ap); return Write(buf, len); }
};
