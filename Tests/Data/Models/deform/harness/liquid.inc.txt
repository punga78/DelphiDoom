// idToken and idParser are stand-ins: this harness has a minimal idStr (see precompiled.h.txt), which the real
// Token.h and Lexer.h cannot use. They split the text at white space (a quoted string is one token, without the
// quotes) and read numbers with atoi and atof; the lexical details of the real parser have their own parity
// (Tests\Data\Decl, Libs\Scanner). What is real here is every line of Model_liquid.cpp.
extern char g_parserWarning[512];
class idToken {
public:
	std::string s;
	int Icmp(const char *c) const { std::string a = s, b = c; for (size_t i = 0; i < a.size(); i++) a[i] = (char)tolower((unsigned char)a[i]); for (size_t i = 0; i < b.size(); i++) b[i] = (char)tolower((unsigned char)b[i]); return a.compare(b); }
	const char *c_str() const { return s.c_str(); }
	operator const char *() const { return s.c_str(); }
};
class idParser {
public:
	std::vector<std::string> tokens; size_t at; std::string fileName;
	idParser(int flags) : at(0) {}
	bool LoadMemory(const char *text, int length, const char *name) {
		fileName = name; std::string t(text, length); size_t i = 0;
		while (i < t.size()) {
			if (isspace((unsigned char)t[i])) { i++; continue; }
			if (t[i] == '"') { size_t j = t.find('"', i + 1); tokens.push_back(t.substr(i + 1, j - i - 1)); i = j + 1; }
			else { size_t j = i; while (j < t.size() && !isspace((unsigned char)t[j])) j++; tokens.push_back(t.substr(i, j - i)); i = j; }
		}
		return true;
	}
	bool ReadToken(idToken *token) { if (at >= tokens.size()) return false; token->s = tokens[at++]; return true; }
	int ParseInt() { return at < tokens.size() ? atoi(tokens[at++].c_str()) : 0; }
	float ParseFloat() { return at < tokens.size() ? (float)atof(tokens[at++].c_str()) : 0.0f; }
	void Warning(const char *fmt, ...) { va_list a; va_start(a, fmt); vsprintf(g_parserWarning, fmt, a); va_end(a); }
};

static void F2(float f) { unsigned u; memcpy(&u, &f, 4); printf(" #%08x", u); }

static void PrintSurface(const char *label, int k, int t, srfTriangles_t *tri, bool full) {
	printf("LIQ %s t%d | %d %d %d\n", label, k, t, tri->numVerts, tri->numIndexes);
	printf("LIQ %s t%d z |", label, k); for (int i = 0; i < tri->numVerts; i++) F2(tri->verts[i].xyz.z); printf("\n");
	printf("LIQ %s t%d bounds |", label, k); for (int a = 0; a < 2; a++) { F2(tri->bounds[a].x); F2(tri->bounds[a].y); F2(tri->bounds[a].z); } printf("\n");
	if (full) {
		printf("LIQ %s t%d normals |", label, k);
		for (int i = 0; i < tri->numVerts; i++) { F2(tri->verts[i].normal.x); F2(tri->verts[i].normal.y); F2(tri->verts[i].normal.z); F2(tri->verts[i].tangents[0].x); F2(tri->verts[i].tangents[0].y); F2(tri->verts[i].tangents[0].z); F2(tri->verts[i].tangents[1].x); F2(tri->verts[i].tangents[1].y); F2(tri->verts[i].tangents[1].z); }
		printf("\n");
	}
}

// A case: the text of the file, the times, the times that print the normals and tangents too, and an optional
// IntersectBounds before the time index "intersectAt" (-1: none)
static void LiquidCase(const char *label, const char *text, const int *times, int numTimes, int fullMask, int intersectAt) {
	g_liquidText = text;
	idRenderModelLiquid *m = new idRenderModelLiquid;
	m->deformInfo = 0;
	g_parserWarning[0] = 0;
	m->InitFromFile("test.liquid");
	if (g_parserWarning[0]) { printf("LIQ %s warning | %s\n", label, g_parserWarning); g_parserWarning[0] = 0; }
	printf("LIQ %s text |", label); for (const char *p = text; *p; p++) { if (*p == '\n') putchar(' '); else putchar(*p); } printf("\n");
	printf("LIQ %s init | %d %d %d %d %d\n", label, m->defaulted ? 1 : 0, m->verts_x, m->verts_y, m->update_tics, m->liquid_type);
	if (m->defaulted) { delete m; return; }
	printf("LIQ %s dims |", label); F2(m->scale_x); F2(m->scale_y); F2(m->density); F2(m->drop_height); F2(m->drop_delay); printf(" %d %d %d\n", m->drop_radius, m->seed, m->deformInfo ? 1 : 0);
	printf("LIQ %s modelbounds |", label); for (int a = 0; a < 2; a++) { F2(m->bounds[a].x); F2(m->bounds[a].y); F2(m->bounds[a].z); } printf("\n");
	printf("LIQ %s shader | %s\n", label, m->shader->name.c_str());
	printf("LIQ %s times |", label); for (int k = 0; k < numTimes; k++) printf(" %d", times[k]); printf("\n");
	for (int k = 0; k < numTimes; k++) {
		if (k == intersectAt) {
			idBounds b; b[0] = idVec3(m->scale_x * 2.5f, m->scale_y * 3.5f, -2.0f); b[1] = idVec3(m->scale_x * 6.5f, m->scale_y * 7.25f, 1.5f);
			m->IntersectBounds(b, 0.0f);
			printf("LIQ %s intersect | %d", label, k); for (int a = 0; a < 2; a++) { F2(b[a].x); F2(b[a].y); F2(b[a].z); } printf("\n");
		}
		viewDef_s view; view.renderView.time = times[k];
		idRenderModel *r = m->InstantiateDynamicModel(0, &view, 0);
		PrintSurface(label, k, times[k], r->surfaces[0].geometry, ((fullMask >> k) & 1) != 0);
		printf("LIQ %s t%d state | %d %d\n", label, k, m->time, m->nextDropTime);
	}
	delete m;
}
