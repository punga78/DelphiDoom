enum fsOrigin_t { FS_SEEK_CUR, FS_SEEK_END, FS_SEEK_SET };
class idFile {
public:
	std::string rd; size_t pos; long ts;
	idFile() : pos(0), ts(0) {}
	int Read(void *b, int l) { if (pos + l > rd.size()) l = (int)(rd.size() - pos); memcpy(b, rd.data() + pos, l); pos += l; return l; }
	int Seek(long offset, fsOrigin_t origin) { if (origin == FS_SEEK_CUR) pos += offset; else if (origin == FS_SEEK_SET) pos = offset; else pos = rd.size() + offset; return 0; }
	int Tell() { return (int)pos; }
	long Timestamp() { return ts; }
};
