void BigRevBytes(void *bp, int elsize, int elcount) {
	unsigned char *p = (unsigned char *)bp;
	for (int i = 0; i < elcount; i++, p += elsize) for (int j = 0; j < elsize / 2; j++) { unsigned char t = p[j]; p[j] = p[elsize - 1 - j]; p[elsize - 1 - j] = t; }
}

static void hex(float f) { unsigned int u; memcpy(&u, &f, 4); printf(" #%08x", u); }

static void dump(idRenderModelStatic &m, bool ok) {
	printf("%s result | %d\n", label.c_str(), ok ? 1 : 0);
	printf("%s surfaces | %d\n", label.c_str(), m.NumSurfaces());
	for (int s = 0; s < m.NumSurfaces(); s++) {
		modelSurface_t &ms = m.surfaces[s];
		srfTriangles_t *t = ms.geometry;
		printf("%s s%d info | %d %s\n", label.c_str(), s, ms.id, ((const idMaterial *)ms.shader)->name.c_str());
		if (!t) { printf("%s s%d null | 1\n", label.c_str(), s); continue; }
		printf("%s s%d counts | %d %d %d\n", label.c_str(), s, t->numVerts, t->numIndexes, t->generateNormals ? 1 : 0);
		printf("%s s%d indexes |", label.c_str(), s);
		for (int i = 0; i < t->numIndexes; i++) printf(" %d", t->indexes[i]);
		printf("\n");
		float mn[3] = { 1e30f, 1e30f, 1e30f }, mx[3] = { -1e30f, -1e30f, -1e30f };
		for (int i = 0; i < t->numVerts; i++) {
			idDrawVert &v = t->verts[i];
			printf("%s s%d v%d |", label.c_str(), s, i);
			hex(v.xyz.x); hex(v.xyz.y); hex(v.xyz.z); hex(v.st.x); hex(v.st.y); hex(v.normal.x); hex(v.normal.y); hex(v.normal.z);
			printf(" %d %d %d %d\n", v.color[0], v.color[1], v.color[2], v.color[3]);
			for (int k = 0; k < 3; k++) { if (v.xyz[k] < mn[k]) mn[k] = v.xyz[k]; if (v.xyz[k] > mx[k]) mx[k] = v.xyz[k]; }
		}
		printf("%s s%d bounds |", label.c_str(), s);
		for (int k = 0; k < 3; k++) hex(mn[k]);
		for (int k = 0; k < 3; k++) hex(mx[k]);
		printf("\n");
	}
}

int main(int argc, char **argv) {
	idMath::Init(); idCommon theCommon; common = &theCommon; idLib::common = &theCommon;
	idFileSystem theFs; fileSystem = &theFs; idLib::fileSystem = &theFs;
	tr.defaultMaterial = declManager->FindMaterial("_default");
	std::ifstream jobs(argv[1]);
	std::string kind, path; int merge, fast;
	while (jobs >> label >> kind >> path >> merge >> fast) {
		idRenderModelStatic m;
		m.r_mergeModelSurfaces.f = (float)merge; m.fastLoad = fast != 0;
		bool ok = false;
		try {
			if (kind == "lwo") {
				unsigned int failID = 0; int failpos = 0;
				lwObject *lwo = lwGetObject(path.c_str(), &failID, &failpos);
				if (!lwo) { printf("%s error | lwGetObject failed at %d\n", label.c_str(), failpos); continue; }
				ok = m.ConvertLWOToModelSurfaces(lwo);
			} else {
				aseModel_t *ase = ASE_Load(path.c_str());
				if (!ase) { printf("%s error | ASE_Load failed\n", label.c_str()); continue; }
				ok = m.ConvertASEToModelSurfaces(ase);
			}
		} catch (idException &e) { printf("%s error | %s\n", label.c_str(), e.error); continue; }
		dump(m, ok);
	}
	return 0;
}
