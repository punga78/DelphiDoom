class idImageManager {
public:
	idImage *defaultImage, *currentRenderImage;
	idIntCVar image_ignoreHighQuality;
	idImageManager() { defaultImage = new idImage; defaultImage->imgName = "_default"; currentRenderImage = new idImage; currentRenderImage->imgName = "_currentRender"; }
	idImage *ImageFromFile( const char *_name, textureFilter_t filter, bool allowDownSize, textureRepeat_t repeat, textureDepth_t depth, cubeFiles_t cubeMap = CF_2D ) {
		if ( !_name || !_name[0] || idStr::Icmp( _name, "default" ) == 0 || idStr::Icmp( _name, "_default" ) == 0 ) {
			return defaultImage;
		}
		idStr name = _name;
		name.Replace( ".tga", "" );
		name.BackSlashesToSlashes();
		if ( name.Icmp( "_currentRender" ) == 0 ) {
			return currentRenderImage;
		}
		idImage *image = new idImage;
		image->imgName = name; image->filter = filter; image->repeat = repeat; image->depth = depth; image->cube = cubeMap; image->picmip = allowDownSize;
		return image;
	}
};
