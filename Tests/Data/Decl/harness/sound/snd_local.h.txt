class idSoundSample {
public:
	idStr name;
	waveformatex_t objectInfo;
	int objectSize;
	int objectMemSize;
	bool defaultSound;
	// the body of idSoundSample::LengthIn44kHzSamples of snd_cache.cpp
	int LengthIn44kHzSamples( void ) const {
		if ( objectInfo.nSamplesPerSec == 11025 ) return objectSize << 2;
		else if ( objectInfo.nSamplesPerSec == 22050 ) return objectSize << 1;
		else return objectSize << 0;
	}
};
