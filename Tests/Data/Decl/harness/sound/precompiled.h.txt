class idDecl {
public:
	idDecl( void ) : replaced( false ), recursion( 0 ) {}
	virtual ~idDecl( void ) {}
	const char *GetName( void ) const { return name.c_str(); }
	const char *GetFileName( void ) const { return "test.decl"; }
	int GetLineNum( void ) const { return 1; }
	int Index( void ) const { return 0; }
	void SetText( const char *t ) { text = t; }
	bool ReplaceSourceFileText( void ) { replaced = true; return true; }
	void MakeDefault( void ) {
		if ( ++recursion > 100 ) { recursion = 0; common->Error( "recursion" ); }
		FreeData();
		const char *d = DefaultDefinition();
		Parse( d, (int)strlen( d ) );
		--recursion;
	}
	virtual size_t Size( void ) const { return 0; }
	virtual bool SetDefaultText( void ) { return false; }
	virtual const char *DefaultDefinition( void ) const { return "{ }"; }
	virtual bool Parse( const char *t, const int l ) { return true; }
	virtual void FreeData( void ) {}
	virtual void List( void ) const {}
	virtual void Print( void ) const {}
	std::string name, text;
	bool replaced;
	int recursion;
};
