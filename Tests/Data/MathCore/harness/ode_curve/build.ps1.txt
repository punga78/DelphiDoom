# Builds the harness of ode.golden.txt and curve.golden.txt (see Libs\MathCore\CLAUDE.md, issue #95).
# $d has idlib\ (a copy of neo\idlib) with these files: driver.cpp (driver.cpp.txt) and precompiled.h (precompiled.h.txt).
# Only change to the idlib sources: in math\Ode.cpp the four constructors take "const deriveFunction_t dr" (as Ode.h declares):
# bcc32c mangles the top-level const and does not link the original definitions.
$d = 'C:\harness\ode_curve'
Set-Location "$d\idlib"
New-Item -ItemType Directory -Force "$d\obj" | Out-Null
$B = 'C:\Program Files (x86)\Embarcadero\Studio\22.0\bin\bcc32c.exe'
$files = 'driver','Str','Heap','math\Math','math\Vector','math\Matrix','math\Angles','math\Quat','math\Rotation','math\Plane','math\Pluecker','math\Polynomial','math\Complex','math\Simd_Generic','math\Ode','geometry\JointTransform'
foreach ($f in $files) {
  $o = "$d\obj\" + ($f -replace '\\','_') + '.o'
  $out = & $B -c -w- -std=c++14 -I. "-o$o" "$f.cpp" 2>&1
  if ($LASTEXITCODE -ne 0) { Write-Host "== $f exit $LASTEXITCODE"; $out | Select-String 'error' | Select-Object -First 6 | ForEach-Object { $_.Line } }
}
$all = Get-ChildItem "$d\obj\*.o" | ForEach-Object { $_.FullName }
& $B -o "$d\oc.exe" @all
& "$d\oc.exe" ode > "$d\ode.txt"
& "$d\oc.exe" curve > "$d\curve.txt"
