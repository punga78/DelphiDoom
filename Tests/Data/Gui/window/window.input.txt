# Scenarios of the .gui of Doom 3 (issue #111), written for the test: no file of Doom 3.
# @case name | @file path (the lines up to the next directive) | @op operation | @end
# operations: load FILE, frame T, state KEY VALUE, statechanged T, named NAME, activate 0|1 T, trigger T, dump
@case basic
@file basic.gui
windowDef Desktop {
  rect 0, 0, 640, 480
  backcolor 0.1, 0.2, 0.3, 1
  text "Hello"
  windowDef plain {
    rect 10, 20, 100, 50
    forecolor 1, 0, 0, 1
    text "plain"
    notime 1
    bordersize 2
  }
  windowDef action {
    rect 5, 5, 20, 20
    onAction { set "cmd" "go" }
  }
  windowDef shown {
    rect 0, 0, 10, 10
    visible time > 40
  }
  windowDef label {
    rect 1, 2, 3, 4
    matcolor 1, 1, 1, 0.5
    textscale 0.5
    textalign 1
    textalignx 3
    font "fonts/a"
    shadow 2
    comment "a note"
  }
}
@op load basic.gui
@op dump
@op frame 100
@op dump
@end
@case expressions
@file expr.gui

  rect 0, 0, time / 10, 20 + 5 * 2
  backcolor 1, 0, 0, time > 50 ? 1 : 0.5
  visible time >= 100 && time < 200
  windowDef calc {
    rect 17 % 5, 10 - 4 - 3, (2 + 3) * 4, 5 / 0
    forecolor "gui::r", "gui::g", 0, 1
    text "gui::label"
    hovercolor time % 7, 1, 1, 1
  }
  windowDef later {
    rect later2 * 2, 0, 0, 0
  }
  float later2 4
}

@case scripts
@file scripts.gui
windowDef Desktop {
  rect 0, 0, 640, 480
  text "start"
  onActivate {
    set "text" "activated" ;
    set "cmd" "quit" ;
    set "cmd" "play" "snd x" "y" ;
    runScript "go" ;
    if ( "gui::flag" == 1 ) { set "cmd" "yes" } else { set "cmd" "no" }
  }
  onDeactivate { set "cmd" "off" }
  onTime 100 { set "cmd" "late" }
  onTime 0 { set "cmd" "early" }
  onNamedEvent ping { set "cmd" "pong" }
  onTrigger { set "cmd" "trig" }
  windowDef kid {
    rect 1, 1, 2, 2
    onAction { }
    onNamedEvent ping { set "cmd" "kid" }
  }
}
@op load scripts.gui
@op state flag 1
@op frame 16
@op activate 1 20
@op dump
@op frame 130
@op dump
@op named ping
@op dump
@op trigger 140
@op dump
@op state flag 0
@op frame 200
@op activate 1 210
@op dump
@op activate 0 220
@op dump
@end
