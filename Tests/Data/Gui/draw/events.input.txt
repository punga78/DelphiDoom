# scenarios of the events of the windows of a .gui of Doom 3 (written for the test; no file of Doom 3)
# "event <type> <value> <value2> <time>": 1 key (value 187 is the left mouse button, value2 1 down and 0 up), 2 character,
# 3 mouse movement; the line E is the command that HandleEvent returns, whether the visuals changed and the cursor.
@case events_mouse_and_clicks
@file mouse.gui
windowDef Desktop {
  rect 0, 0, 640, 480
  onAction { set "cmd" "desktop click" }
  onActionRelease { set "cmd" "desktop release" }
  windowDef left {
    rect 10, 10, 100, 100
    backcolor 1, 0, 0, 1
    onMouseEnter { set "cmd" "enter left" }
    onMouseExit { set "cmd" "exit left" }
    onAction { set "cmd" "click left" }
    onActionRelease { set "cmd" "release left" }
  }
  windowDef right {
    rect 200, 10, 100, 100
    backcolor 0, 1, 0, 1
    onMouseEnter { set "cmd" "enter right" }
    onMouseExit { set "cmd" "exit right" }
    onAction { set "cmd" "click right" }
  }
  windowDef covered {
    rect 20, 20, 30, 30
    backcolor 0, 0, 1, 1
    onAction { set "cmd" "click covered" }
  }
  windowDef deaf {
    rect 400, 10, 100, 100
    noevents 1
    onAction { set "cmd" "never" }
  }
  windowDef gone {
    rect 400, 200, 100, 100
    visible 0
    onAction { set "cmd" "never" }
  }
}
@op load mouse.gui
@op redraw 10
@op event 3 50 50 20
@op event 3 5 5 30
@op event 1 187 1 40
@op event 1 187 0 50
@op event 3 150 -30 60
@op event 3 100 0 70
@op event 1 187 1 80
@op event 1 187 0 90
@op event 3 150 5 100
@op event 3 5 5 110
@op event 3 -1000 -1000 120
@op event 1 187 1 130
@op event 1 187 0 140
@op event 3 600 100 150
@op event 1 187 1 160
@op event 1 187 0 170
@op event 0 0 0 180
@end
@case events_focus_and_tab
@file tab.gui
windowDef Desktop {
  rect 0, 0, 640, 480
  wantenter 1
  onAction { set "cmd" "desktop action" }
  onActionRelease { set "cmd" "desktop action release" }
  onESC { set "cmd" "desktop esc" }
  windowDef one {
    rect 10, 10, 50, 50
    onAction { set "cmd" "one" }
  }
  windowDef two {
    rect 100, 10, 50, 50
    onAction { set "cmd" "two" }
  }
  windowDef group {
    rect 200, 10, 200, 200
    onMouseEnter { set "cmd" "group enter" }
    windowDef inner1 {
      rect 10, 10, 50, 50
      onAction { set "cmd" "inner1" }
    }
    windowDef inner2 {
      rect 100, 10, 50, 50
      onAction { set "cmd" "inner2" }
    }
  }
  windowDef three {
    rect 10, 300, 50, 50
    onAction { set "cmd" "three" }
  }
}
@op load tab.gui
@op redraw 10
@op event 1 9 1 20
@op event 3 20 20 30
@op event 1 187 1 40
@op event 1 187 0 50
@op event 1 9 1 60
@op event 1 9 1 70
@op dump
@op event 1 9 1 80
@op event 1 9 1 90
@op event 1 9 1 100
@op dump
@op event 1 9 1 110
@op keydown 142
@op event 1 9 1 120
@op event 1 9 1 130
@op dump
@op keyup 142

@op event 1 27 1 150
@op event 1 27 0 160
@op event 1 13 1 170
@op event 1 13 0 180
@op event 1 65 1 190

@end
@case events_pending_and_state
